from pathlib import Path

input_root = Path("/mnt/8EEE3D67EE3D492B/terreus_genotyping/results/0_6_antismash/unmod")
output_root = Path("/mnt/8EEE3D67EE3D492B/terreus_genotyping/results/0_6_antismash/mod")

for gbk in input_root.rglob("*.gbk"):
    # genome directory is the directory two levels above the GBK:
    # genome_1/results/file.gbk
    genome_name = gbk.parent.parent.name

    # Preserve directory structure in the output
    relative_path = gbk.relative_to(input_root)
    output_file = output_root / relative_path
    output_file.parent.mkdir(parents=True, exist_ok=True)

    with open(gbk, "r") as f:
        text = f.read()

    old = "ORGANISM  Aspergillus terreus Unclassified."
    new = f"ORGANISM  Aspergillus terreus {genome_name}"

    if old not in text:
        print(f"WARNING: pattern not found in {gbk}")
    else:
        text = text.replace(old, new)

    with open(output_file, "w") as f:
        f.write(text)

    print(f"{gbk} -> {new}")

def linkage_to_newick(linked, labels=None):
    """
    Convert scipy linkage matrix to Newick tree string.
    Parameters:
        linked : array-like, linkage matrix from scipy.cluster.hierarchy
        labels : list of strings (optional) – leaf labels. If None, uses 'i' as name.
    Returns:
        newick string
    """
    n = linked.shape[0] + 1  # number of leaves
    
    # Use labels if provided, else fallback to str(i)
    if labels is not None:
        node_name = {i: labels[i] for i in range(n)}
    else:
        node_name = {i: str(i) for i in range(n)}

    for i, row in enumerate(linked):
        a, b, dist, size = row
        a, b = int(a), int(b)
        node_name[i + n] = f"({node_name[a]}:{dist/2:.4f},{node_name[b]}:{dist/2:.4f})"

    root = 2 * n - 2  # last internal node index
    return node_name[root] + ";"
