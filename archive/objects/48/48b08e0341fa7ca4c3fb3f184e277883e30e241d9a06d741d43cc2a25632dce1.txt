for dir in /mnt/8EEE3D67EE3D492B/terreus_genotyping/results2/0_1_spades/terreus/*/; do
	base=$(basename $dir)
	if [[ ! -f "/mnt/8EEE3D67EE3D492B/terreus_genotyping/results2/0_2_repeat_mask_funannotate/$base/ater/annotate_results/Aspergillus_terreus.transcripts.fa" ]]; then
		outdir=/mnt/8EEE3D67EE3D492B/terreus_genotyping/results2/0_2_repeat_mask_funannotate
		mkdir -p "$outdir/$base/ater"
		conda run -n seqkit seqkit replace -p ".+" -r "contig_{nr}" $dir/contigs.fasta > $outdir/$base/contigs_renamed.fasta
		conda run -n repeats BuildDatabase -name $outdir/$base/repeats_db $outdir/$base/contigs_renamed.fasta
		conda run -n repeats RepeatModeler -database $outdir/$base/repeats_db -threads 16
		conda run -n repeats RepeatMasker -pa 16 -lib $outdir/$base/repeats_db-families.fa -xsmall $outdir/$base/contigs_renamed.fasta
		conda run -n funannotate2 funannotate2 clean -f $outdir/$base/contigs_renamed.fasta.masked -o $outdir/$base/cleaned_masked_contigs.fasta
		conda run -n funannotate2 funannotate2 train -f $outdir/$base/cleaned_masked_contigs.fasta -s "Aspergillus terreus" -o $outdir/$base/ater
		conda run -n funannotate2 funannotate2 predict -i $outdir/$base/ater
		conda run -n funannotate2 funannotate2 annotate -i $outdir/$base/ater
	fi
done
