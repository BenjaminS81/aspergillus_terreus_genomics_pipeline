mkdir /mnt/8EEE3D67EE3D492B/terreus_genotyping/results/3_1_lov_cluster/terreus2/lvrA_HMDH
for dir in /mnt/8EEE3D67EE3D492B/terreus_genotyping/results/0_2_repeat_mask_funannotate/terreus/*; do
    base=$(basename "$dir")
    tmp=$(mktemp)
    awk '
    BEGIN {FS="\t"; IGNORECASE=1}
    $3=="mRNA" && tolower($9) ~ /HMDH/ {
        if (match($9,/ID=[^;]+/))
            print substr($9,RSTART+3,RLENGTH-3)
    }
    ' "$dir/ater/annotate_results/Aspergillus_terreus.gff3" > "$tmp"
    if [ -s "$tmp" ]; then
        conda run -n seqkit seqkit grep -f "$tmp" "$dir/ater/annotate_results/Aspergillus_terreus.transcripts.fa" \
        | sed "s/^>/>${base}|/"
    fi
    rm "$tmp"
done >> /mnt/8EEE3D67EE3D492B/terreus_genotyping/results/3_1_lov_cluster/terreus2/lvrA_HMDH/lvrA_HMDH.fa


conda run -n mafft mafft --inputorder --adjustdirectionaccurately /mnt/8EEE3D67EE3D492B/terreus_genotyping/results/3_1_lov_cluster/lovF/lovF.fa > /mnt/8EEE3D67EE3D492B/terreus_genotyping/results/3_1_lov_cluster/lovF/lovF_mafft.fa
